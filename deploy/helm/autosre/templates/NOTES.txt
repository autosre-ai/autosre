1. Get the application URLs:
{{- if .Values.ingress.enabled }}

  Web UI:   https://{{ .Values.ingress.hosts.web }}
  API:      https://{{ .Values.ingress.hosts.api }}

{{- else if .Values.web.enabled }}

  Get the Web UI URL by running:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "autosre.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=web" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:3000 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 3000:{{ .Values.web.service.port }}

{{- end }}
{{- if .Values.api.enabled }}

  Get the API URL by running:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "autosre.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=api" -o jsonpath="{.items[0].metadata.name}")
  echo "API available at http://127.0.0.1:8000"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8000:{{ .Values.api.service.port }}

{{- end }}

2. Check deployment status:

  kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/instance={{ .Release.Name }}"

3. View logs:

  # Agent logs
  kubectl logs -f --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "autosre.name" . }},app.kubernetes.io/component=agent"

  # API logs
  kubectl logs -f --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "autosre.name" . }},app.kubernetes.io/component=api"
